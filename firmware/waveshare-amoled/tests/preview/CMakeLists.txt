cmake_minimum_required(VERSION 3.16)
project(bedside_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
add_executable(preview preview.cpp ../../main/ui.cpp ../../main/model.cpp ../../main/bedroom.cpp ../../main/bed_scene.cpp ../../main/temperature.cpp "$ENV{IDF_PATH}/components/json/cJSON/cJSON.c")
target_include_directories(preview PRIVATE . ../../main "$ENV{IDF_PATH}/components/json/cJSON")
target_link_libraries(preview PRIVATE lvgl)
