cmake_minimum_required(VERSION 3.16)
project(bedside_model_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(JSON_SOURCE "$ENV{IDF_PATH}/components/json/cJSON")
add_executable(model_tests model_tests.cpp ../main/model.cpp ../main/personal.cpp ../main/bedroom.cpp ../main/bed_scene.cpp ../main/temperature.cpp "${JSON_SOURCE}/cJSON.c")
target_include_directories(model_tests PRIVATE ../main "${JSON_SOURCE}")
target_compile_options(model_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_test(NAME bedside_models COMMAND model_tests)
add_executable(preferences_tests preferences_tests.cpp ../main/preferences.cpp)
target_include_directories(preferences_tests PRIVATE ../main fakes)
target_compile_options(preferences_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME display_preferences COMMAND preferences_tests)
add_executable(bedroom_tests bedroom_tests.cpp ../main/model.cpp ../main/personal.cpp ../main/bedroom.cpp "${JSON_SOURCE}/cJSON.c")
target_include_directories(bedroom_tests PRIVATE ../main "${JSON_SOURCE}")
target_compile_definitions(bedroom_tests PRIVATE BEDROOM_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/fixtures/bedroom-core.json")
target_compile_options(bedroom_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME bedroom_contract COMMAND bedroom_tests)
add_executable(personal_tests personal_tests.cpp ../main/model.cpp ../main/personal.cpp ../main/bedroom.cpp "${JSON_SOURCE}/cJSON.c")
target_include_directories(personal_tests PRIVATE ../main "${JSON_SOURCE}")
target_compile_definitions(personal_tests PRIVATE PERSONAL_FIXTURE_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
target_compile_options(personal_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME personal_remote_contract COMMAND personal_tests)
